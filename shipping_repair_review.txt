 147: def get_ups_access_token():
 148:     """
 149:     Request a fresh UPS OAuth client-credentials token.
 150: 
 151:     Call only after UPS has supplied valid API credentials.
 152:     Tokens are deliberately not cached or written to disk.
 153:     """
 154:     config = get_ups_config()
 155: 
 156:     if not config["client_id"] or not config["client_secret"]:
 157:         raise RuntimeError(
 158:             "UPS credentials are missing. Set UPS_CLIENT_ID and "
 159:             "UPS_CLIENT_SECRET before testing authentication."
 160:         )
 161: 
 162:     token_url = config["api_base"] + "/security/v1/oauth/token"
 163: 
 164:     credentials = (
 165:         config["client_id"] + ":" + config["client_secret"]
 166:     ).encode("utf-8")
 167: 
 168:     import base64
 169: 
 170:     authorization = base64.b64encode(credentials).decode("ascii")
 171: 
 172:     body = urllib.parse.urlencode({
 173:         "grant_type": "client_credentials"
 174:     }).encode("utf-8")
 175: 
 176:     request = urllib.request.Request(
 177:         token_url,
 178:         data=body,
 179:         headers={
 180:             "Authorization": "Basic " + authorization,
 181:             "Content-Type": "application/x-www-form-urlencoded",
 182:             "Accept": "application/json",
 183:         },
 184:         method="POST",
 185:     )
 186: 
 187:     try:
 188:         with urllib.request.urlopen(
 189:             request, timeout=config["timeout"]
 190:         ) as response:
 191:             result = json.loads(response.read().decode("utf-8"))
 192: 
 193:     except urllib.error.HTTPError as exc:
 194:         raise RuntimeError(
 195:             "UPS OAuth authentication failed with HTTP "
 196:             f"{exc.code}. Verify credentials, environment, and "
 197:             "the UPS application's enabled APIs."
 198:         ) from None
 199: 
 200:     except (urllib.error.URLError, TimeoutError, OSError):
 201:         raise RuntimeError(
 202:             "Could not reach the UPS OAuth service."
 203:         ) from None
 204: 
 205:     except (ValueError, UnicodeDecodeError):
 206:         raise RuntimeError(
 207:             "UPS OAuth returned an invalid response."
 208:         ) from None
 209: 
 210:     token = result.get("access_token")
 211: 
 212:     if not token:
 213:         raise RuntimeError(
 214:             "UPS did not return an access token."
 215:         )
 216: 
 217:     return token
