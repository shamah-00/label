def checkout(request):
    # LABEL_GROUP_US_ONLY_SHIPPING_GUARD
    # This checkout is configured for US domestic shipping only.
    # Normalize the fixed checkout country before downstream validation
    # and Shippo rate requests so inconsistent browser values cannot break checkout.
    if request.method == "POST":
        request.POST = request.POST.copy()
        request.POST["shipping_country"] = "United States"

    import hashlib
    import json
    import os
    import time
    from decimal import Decimal, InvalidOperation

    import requests
    from django.contrib import messages
    from django.db import transaction
    from django.shortcuts import render, redirect

    from .models import Product, Order, OrderItem

    ZERO = Decimal("0.00")
    SESSION_RATES = "shippo_checkout_rates"
    SESSION_SIGNATURE = "shippo_checkout_signature"
    SESSION_TIME = "shippo_checkout_rates_time"

    def render_checkout(items, subtotal, rates=None, error="", shipping_cost=ZERO):
        total = subtotal + shipping_cost
        context = {
            "checkout_items": items,
            "cart_items": items,
            "items": items,
            "subtotal": subtotal,
            "cart_total": subtotal,
            "shipping_cost": shipping_cost,
            "total": total,
            "grand_total": total,
            "shipping_rates": rates or [],
            "shipping_rate_error": error,
            "rate_error": error,
        }
        return render(request, "checkout.html", context)

    def get_items():
        items = []
        cart = request.session.get("cart", {})

        if not isinstance(cart, dict):
            return items

        for product_id, raw_quantity in cart.items():
            try:
                quantity = int(raw_quantity)
                if quantity < 1:
                    continue

                product = Product.objects.get(
                    pk=int(product_id),
                    is_available=True,
                )

                unit_price = (
                    product.discount_price
                    if product.discount_price is not None
                    and product.discount_price > ZERO
                    else product.price
                )
                unit_price = Decimal(str(unit_price))
                line_total = unit_price * quantity

                items.append({
                    "product": product,
                    "quantity": quantity,
                    "unit_price": unit_price,
                    "subtotal": line_total,
                    "total_price": line_total,
                    "item_total": line_total,
                })
            except (
                Product.DoesNotExist,
                ValueError,
                TypeError,
                InvalidOperation,
            ):
                continue

        return items

    def country_code(value):
        value = (value or "").strip()
        if len(value) == 2 and value.isalpha():
            return value.upper()

        aliases = {
            "UNITED STATES": "US",
            "UNITED STATES OF AMERICA": "US",
            "USA": "US",
            "UNITED KINGDOM": "GB",
            "GREAT BRITAIN": "GB",
            "KENYA": "KE",
            "PHILIPPINES": "PH",
            "CANADA": "CA",
            "AUSTRALIA": "AU",
            "NEW ZEALAND": "NZ",
            "SOUTH AFRICA": "ZA",
            "INDIA": "IN",
            "UGANDA": "UG",
            "TANZANIA": "TZ",
        }
        return aliases.get(value.upper(), "")

    def build_packages(items):
        packages = []
        missing = []

        for item in items:
            product = item["product"]
            quantity = item["quantity"]

            try:
                weight = Decimal(str(product.shipping_weight_lb))
                length = Decimal(str(product.shipping_length_in))
                width = Decimal(str(product.shipping_width_in))
                height = Decimal(str(product.shipping_height_in))

                if any(v <= 0 for v in (weight, length, width, height)):
                    raise ValueError("measurements must be greater than zero")

            except (ValueError, TypeError, InvalidOperation):
                missing.append(
                    "{} ({})".format(
                        product.name,
                        getattr(product, "product_code", product.pk),
                    )
                )
                continue

            # Each unit is represented as a parcel. This deliberately avoids
            # inventing combined-package dimensions for multiple quantities.
            for unused in range(quantity):
                packages.append({
                    "length": str(length),
                    "width": str(width),
                    "height": str(height),
                    "distance_unit": "in",
                    "weight": str(weight),
                    "mass_unit": "lb",
                })

        if missing:
            raise ValueError(
                "Shipping measurements are missing or invalid for: "
                + ", ".join(missing[:12])
                + (
                    " and additional products"
                    if len(missing) > 12 else ""
                )
                + ". Enter the actual packed weight and outside dimensions "
                  "for each product, then calculate shipping again."
            )

        if not packages:
            raise ValueError("There are no valid packages to ship.")

        if len(packages) > 100:
            raise ValueError(
                "This order contains more than 100 individual parcels. "
                "Please request a shipping quote so the packages can be "
                "consolidated and priced correctly."
            )

        return packages

    def checkout_signature(address, items, packages):
        payload = {
            "address": address,
            "cart": [
                {
                    "product_id": item["product"].pk,
                    "quantity": item["quantity"],
                    "unit_price": str(item["unit_price"]),
                }
                for item in items
            ],
            "packages": packages,
        }
        encoded = json.dumps(
            payload,
            sort_keys=True,
            separators=(",", ":"),
        ).encode("utf-8")
        return hashlib.sha256(encoded).hexdigest()

    def get_configured_shipping_rates(address, packages, order_subtotal):
        """
        Calculate store-configured shipping estimates.
        This is NOT a live UPS or Shippo rate.
        """
        from decimal import Decimal, InvalidOperation
        from .models import ShippingRule, ShippingSystemSettings

        if not packages:
            raise ValueError("No valid packages were provided for shipping.")

        settings = ShippingSystemSettings.objects.first()
        if not settings or not settings.enabled:
            raise ValueError(
                "Shipping is currently unavailable. Please contact us "
                "for a shipping quote."
            )

        total_weight = sum(
            (Decimal(str(package["weight"])) for package in packages),
            Decimal("0.00"),
        )
        subtotal_value = Decimal(str(order_subtotal))
        country = str(address.get("country", "")).strip().upper()
        state = str(address.get("state", "")).strip().upper()
        postal = str(address.get("zip", "")).strip().upper()

        def normalized_country(value):
            value = str(value or "").strip().upper()
            aliases = {
                "US": "US",
                "USA": "US",
                "UNITED STATES": "US",
                "UNITED STATES OF AMERICA": "US",
            }
            return aliases.get(value, value)

        matching_rules = []
        for rule in ShippingRule.objects.filter(active=True).order_by("-priority", "id"):
            rule_country = normalized_country(rule.country)
            if rule_country and rule_country != normalized_country(country):
                continue

            rule_state = str(rule.state or "").strip().upper()
            if rule_state and rule_state != state:
                continue

            prefix = str(rule.postal_prefix or "").strip().upper()
            if prefix and not postal.startswith(prefix):
                continue

            minimum = Decimal(str(rule.minimum_weight_lb or "0"))
            if total_weight < minimum:
                continue

            if rule.maximum_weight_lb is not None:
                maximum = Decimal(str(rule.maximum_weight_lb))
                if total_weight > maximum:
                    continue

            matching_rules.append(rule)

        rule = matching_rules[0] if matching_rules else None

        if rule:
            flat_fee = Decimal(str(rule.flat_fee))
            per_lb_fee = Decimal(str(rule.per_lb_fee))
            handling_fee = Decimal(str(rule.handling_fee))
            free_threshold = (
                Decimal(str(rule.free_shipping_threshold))
                if rule.free_shipping_threshold is not None else None
            )
            rate_id = "tlg-configured-rule-{}".format(rule.pk)
            service_name = rule.name or "Standard Shipping"
        else:
            flat_fee = Decimal(str(settings.default_flat_fee))
            per_lb_fee = Decimal(str(settings.default_per_lb_fee))
            handling_fee = Decimal(str(settings.default_handling_fee))
            free_threshold = (
                Decimal(str(settings.default_free_shipping_threshold))
                if settings.default_free_shipping_threshold is not None else None
            )
            rate_id = "tlg-configured-default"
            service_name = "Standard Shipping (Store Estimate)"

        if free_threshold is not None and free_threshold > 0 and subtotal_value >= free_threshold:
            amount = Decimal("0.00")
            service_name += " - Free"
        else:
            amount = flat_fee + (per_lb_fee * total_weight) + handling_fee

        amount = amount.quantize(Decimal("0.01"))

        return [{
            "object_id": rate_id,
            "provider": "The Label Group (Configured Estimate)",
            "service": service_name,
            "amount": str(amount),
            "currency": "USD",
            "estimated_days": None,
        }]

    items = get_items()
    subtotal = sum(
        (item["subtotal"] for item in items),
        ZERO,
    )

    if not items:
        messages.error(
            request,
            "Your cart is empty or its products are unavailable.",
        )
        return redirect("cart")

    if request.method != "POST":
        return render_checkout(items, subtotal)

    action = request.POST.get("action", "").strip()
    if not action:
        action = (
            "place_order"
            if request.POST.get("shipping_rate_id", "").strip()
            else "get_rates"
        )

    customer_name = request.POST.get("customer_name", "").strip()
    customer_email = request.POST.get("customer_email", "").strip()
    customer_phone = request.POST.get("customer_phone", "").strip()
    street = request.POST.get("shipping_address", "").strip()
    city = request.POST.get("shipping_city", "").strip()
    state = request.POST.get("shipping_state", "").strip()
    postal_code = request.POST.get("shipping_postal_code", "").strip()
    country = country_code(request.POST.get("shipping_country", ""))
    notes = request.POST.get("notes", "").strip()

    required = {
        "full name": customer_name,
        "email address": customer_email,
        "phone number": customer_phone,
        "street address": street,
        "city": city,
        "postal code": postal_code,
        "country": country,
    }
    missing_fields = [label for label, value in required.items() if not value]

    if missing_fields:
        return render_checkout(
            items,
            subtotal,
            error="Please complete: " + ", ".join(missing_fields) + ".",
        )

    address = {
        "name": customer_name,
        "street1": street,
        "city": city,
        "state": state,
        "zip": postal_code,
        "country": country,
        "phone": customer_phone,
        "email": customer_email,
    }

    try:
        packages = build_packages(items)
    except ValueError as exc:
        request.session.pop(SESSION_RATES, None)
        request.session.pop(SESSION_SIGNATURE, None)
        request.session.pop(SESSION_TIME, None)
        return render_checkout(items, subtotal, error=str(exc))

    signature = checkout_signature(address, items, packages)
    cached_rates = request.session.get(SESSION_RATES, [])
    cached_signature = request.session.get(SESSION_SIGNATURE)
    cached_time = request.session.get(SESSION_TIME, 0)
    cache_valid = (
        cached_signature == signature
        and time.time() - float(cached_time or 0) <= 900
        and isinstance(cached_rates, list)
        and bool(cached_rates)
    )

    if action == "get_rates":
        try:
            rates = get_configured_shipping_rates(address, packages, subtotal)
        except ValueError as exc:
            request.session.pop(SESSION_RATES, None)
            request.session.pop(SESSION_SIGNATURE, None)
            request.session.pop(SESSION_TIME, None)
            return render_checkout(items, subtotal, error=str(exc))

        if not rates:
            request.session.pop(SESSION_RATES, None)
            request.session.pop(SESSION_SIGNATURE, None)
            request.session.pop(SESSION_TIME, None)
            return render_checkout(
                items,
                subtotal,
                error=(
                    "No shipping services were returned. Check the delivery "
                    "address and package measurements, or request a quote."
                ),
            )

        request.session[SESSION_RATES] = rates
        request.session[SESSION_SIGNATURE] = signature
        request.session[SESSION_TIME] = time.time()
        request.session.modified = True

        return render_checkout(items, subtotal, rates=rates)

    if action != "place_order":
        return render_checkout(
            items,
            subtotal,
            error="Please calculate shipping before placing your order.",
        )

    if not cache_valid:
        request.session.pop(SESSION_RATES, None)
        request.session.pop(SESSION_SIGNATURE, None)
        request.session.pop(SESSION_TIME, None)
        return render_checkout(
            items,
            subtotal,
            error=(
                "Your shipping rates have expired or your address/cart changed. "
                "Calculate shipping again and select a service."
            ),
        )

    selected_id = request.POST.get("shipping_rate_id", "").strip()
    selected_rate = next(
        (
            rate for rate in cached_rates
            if rate.get("object_id") == selected_id
        ),
        None,
    )

    if not selected_rate:
        return render_checkout(
            items,
            subtotal,
            rates=cached_rates,
            error="Select one of the available shipping services.",
        )

    try:
        shipping_cost = Decimal(str(selected_rate["amount"]))
    except (InvalidOperation, KeyError, TypeError):
        return render_checkout(
            items,
            subtotal,
            rates=cached_rates,
            error="The selected shipping price is invalid. Calculate rates again.",
        )

    if shipping_cost < ZERO:
        return render_checkout(
            items,
            subtotal,
            rates=cached_rates,
            error="The selected shipping price is invalid.",
        )

    total = subtotal + shipping_cost
    service_note = (
        "Shipping provider: {}. Service: {}. "
        "Rate reference: {}. Shipping amount: ${} USD."
    ).format(
        selected_rate.get("provider", ""),
        selected_rate.get("service", ""),
        selected_rate.get("object_id", ""),
        format(shipping_cost, ".2f"),
    )
    combined_notes = "\n".join(part for part in (notes, service_note) if part)

    try:
        with transaction.atomic():
            order = Order.objects.create(
                customer_name=customer_name,
                customer_email=customer_email,
                customer_phone=customer_phone,
                shipping_address=street,
                shipping_city=city,
                shipping_state=state,
                shipping_postal_code=postal_code,
                shipping_country=country,
                subtotal=subtotal,
                shipping_cost=shipping_cost,
                total=total,
                currency="USD",
                notes=combined_notes,
            )

            for item in items:
                OrderItem.objects.create(
                    order=order,
                    product=item["product"],
                    product_name=item["product"].name,
                    quantity=item["quantity"],
                    unit_price=item["unit_price"],
                    total_price=item["subtotal"],
                    currency="USD",
                )

    except Exception:
        # Log technical details server-side; don't lose the customer's cart.
        import logging
        logging.getLogger(__name__).exception(
            "Checkout order creation failed."
        )
        return render_checkout(
            items,
            subtotal,
            rates=cached_rates,
            error=(
                "We couldn't save your order. Your cart is preserved. "
                "Please try again or contact us."
            ),
        )

    request.session["cart"] = {}
    request.session.pop(SESSION_RATES, None)
    request.session.pop(SESSION_SIGNATURE, None)
    request.session.pop(SESSION_TIME, None)
    request.session.modified = True

    messages.success(
        request,
        "Your order was submitted successfully. Shipping labels are not "
        "purchased automatically.",
    )
    return redirect("order_success", order_id=order.pk)
