1080: def _label_group_customer_records():
1081:     """
1082:     Build one customer directory from existing orders and quote requests.
1083:     This does not create or modify database records.
1084:     """
1085:     from types import SimpleNamespace
1086:     from products.models import Order, QuoteRequest
1087: 
1088:     records = {}
1089: 
1090:     def value(obj, *names):
1091:         for name in names:
1092:             result = getattr(obj, name, None)
1093:             if result not in (None, ""):
1094:                 return str(result).strip()
1095:         return ""
1096: 
1097:     def add_record(obj, source_type):
1098:         name = value(obj, "customer_name", "full_name", "name")
1099:         email = value(obj, "customer_email", "email").lower()
1100:         phone = value(obj, "customer_phone", "phone", "telephone")
1101:         address = value(obj, "shipping_address", "address")
1102: 
1103:         if not (name or email or phone):
1104:             return
1105: 
1106:         key = email or phone or name.casefold()
1107:         if not key:
1108:             return
1109: 
1110:         if key not in records:
1111:             records[key] = {
1112:                 "name": name or email or phone,
1113:                 "email": email,
1114:                 "phone": phone,
1115:                 "address": address,
1116:                 "total_orders": 0,
1117:                 "total_quotes": 0,
1118:                 "sources": set(),
1119:             }
1120: 
1121:         customer = records[key]
1122: 
1123:         if name and not customer["name"]:
1124:             customer["name"] = name
1125:         if email and not customer["email"]:
1126:             customer["email"] = email
1127:         if phone and not customer["phone"]:
1128:             customer["phone"] = phone
1129:         if address and not customer["address"]:
1130:             customer["address"] = address
1131: 
1132:         customer["sources"].add(source_type)
1133: 
1134:         if source_type == "order":
1135:             customer["total_orders"] += 1
1136:         elif source_type == "quote":
1137:             customer["total_quotes"] += 1
1138: 
1139:     for order in Order.objects.all().iterator():
1140:         add_record(order, "order")
1141: 
1142:     for quote in QuoteRequest.objects.all().iterator():
1143:         add_record(quote, "quote")
1144: 
1145:     customers = []
1146: 
1147:     for item in records.values():
1148:         item["sources"] = ", ".join(sorted(item["sources"]))
1149:         customers.append(SimpleNamespace(
1150:             name=item["name"],
1151:             full_name=item["name"],
1152:             customer_name=item["name"],
1153:             email=item["email"],
1154:             customer_email=item["email"],
1155:             phone=item["phone"],
1156:             customer_phone=item["phone"],
1157:             address=item["address"],
1158:             total_orders=item["total_orders"],
1159:             total_quotes=item["total_quotes"],
1160:             sources=item["sources"],
1161:         ))
1162: 
1163:     return sorted(customers, key=lambda customer: customer.name.lower())

 667: (function () {
 668:     function initCheckout() {
 669:         // Restrict the visible delivery country control to the United States.
 670:         const countryControls = document.querySelectorAll('#shipping_country');
 671: 
 672:         countryControls.forEach(function (control) {
 673:             if (control.tagName === "SELECT") {
 674:                 const options = Array.from(control.options);
 675:                 const usOption = options.find(function (option) {
 676:                     const value = (option.value || "").trim().toLowerCase();
 677:                     const label = (option.textContent || "").trim().toLowerCase();
 678:                     return ["us", "usa", "united states",
 679:                         "united states of america"].includes(value) ||
 680:                         ["us", "usa", "united states",
 681:                         "united states of america"].includes(label);
 682:                 });
 683: 
 684:                 const value = usOption ? usOption.value : "US";
 685:                 control.replaceChildren(new Option("United States", value, true, true));
 686:                 control.value = value;
 687:                 control.required = true;
 688:             } else {
 689:                 control.value = "United States";
 690:                 control.readOnly = true;
 691:                 control.required = true;
 692:             }
 693:         });
 694: 
 695:         // Update shipping and total when the customer chooses a returned rate.
 696:         const totalElement = document.getElementById("checkout-final-total");
 697:         const shippingElement = document.getElementById("checkout-shipping-amount");
 698:         const radios = document.querySelectorAll('input[name="shipping_rate_id"]');
 699: 
 700:         if (!totalElement || !shippingElement) return;
 701: 
 702:         const subtotal = Number.parseFloat(totalElement.dataset.subtotal || "0");
 703:         const money = function (value) {
 704:             return "$" + value.toLocaleString("en-US", {
 705:                 minimumFractionDigits: 2,
 706:                 maximumFractionDigits: 2
 707:             });
 708:         };
 709: 
 710:         function updateTotal() {
 711:             const selected = document.querySelector(
 712:                 'input[name="shipping_rate_id"]:checked'
 713:             );
 714: 
 715:             if (!selected) {
 716:                 shippingElement.textContent = "Select a service below";
 717:                 totalElement.textContent = money(subtotal);
 718:                 return;
 719:             }
 720: 
 721:             const shipping = Number.parseFloat(selected.dataset.amount || "");
 722:             const currency = (selected.dataset.currency || "USD").toUpperCase();
 723: 
 724:             if (!Number.isFinite(shipping) || shipping < 0 || currency !== "USD") {
 725:                 shippingElement.textContent = "Please select a valid USD rate";
 726:                 totalElement.textContent = money(subtotal);
 727:                 return;
 728:             }
 729: 
 730:             shippingElement.textContent = money(shipping) + " USD";
 731:             totalElement.textContent = money(subtotal + shipping);
 732: 
 733:             const help = document.getElementById("shippo-selection-help");
 734:             if (help) {
 735:                 const label = selected.closest("label");
 736:                 const service = label ? label.querySelector(".shippo-rate-info strong") : null;
 737:                 help.textContent = service
 738:                     ? "Selected: " + service.textContent.trim() +
 739:                       ". Your displayed total includes shipping."
 740:                     : "Your displayed total includes shipping.";
 741:             }
 742:         }
 743: 
 744:         radios.forEach(function (radio) {
 745:             radio.addEventListener("change", updateTotal);
 746:         });
 747:         updateTotal();
 748:     }

 751:         document.addEventListener("DOMContentLoaded", initCheckout);
 752:     } else {
